CREATE TABLE customer (
  name TEXT NOT NULL, 
  street_number INT NOT NULL,
  PRIMARY KEY (name, street_number)
);

CREATE TABLE orderr(
  date_of_receipt DATE NOT NULL,
  cname TEXT NOT NULL,
  cstreet_number INT NOT NULL,
  PRIMARY KEY (date_of_receipt, cname, cstreet_number), 
  FOREIGN KEY (cname, cstreet_number) REFERENCES customer(name, street_number)
  ON DELETE CASCADE,
  UNIQUE (date_of_receipt) 
);

CREATE TABLE order_item(
  item_name TEXT NOT NULL,
  item_price INT NOT NULL,
  odate_of_receipt DATE NOT NULL,
  PRIMARY KEY (item_name, odate_of_receipt),  
  FOREIGN KEY (odate_of_receipt) REFERENCES orderr(date_of_receipt)  
  ON DELETE CASCADE
);
